\begin{align} \mathrm{d}\boldsymbol{X} &= \boldsymbol{b}(\boldsymbol{X}, t)\mathrm{d}t + \boldsymbol{B}(\boldsymbol{X}, t)\mathrm{d}\boldsymbol{W}\\ \boldsymbol{X}(0) &= \boldsymbol{X}_{0} \end{align}
